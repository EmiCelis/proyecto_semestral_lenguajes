        function startGame() {
            // Obtener nombres
            const names = [
                document.getElementById('p1-name').value.trim() || 'Jugador 1',
                document.getElementById('p2-name').value.trim() || 'Jugador 2',
                document.getElementById('p3-name').value.trim(),
                document.getElementById('p4-name').value.trim()
            ].filter(n => n);

            if (names.length < 2 || names.length > 4) {
                alert('Deben jugar entre 2 y 4 jugadores.');
                return;
            }

            players = names.map((name, i) => ({
                id: i,
                name,
                coins: 0,
                position: 0,
                joker: false,
                color: ['#ff3333', '#33cc33', '#3366ff', '#ffcc00'][i]
            }));

            // Determinar orden
            players.forEach(p => p.roll = Math.floor(Math.random() * 6) + 1);
            turnOrder = [...players].sort((a, b) => b.roll - a.roll).map(p => p.id);

            // Crear tablero (50 casillas)
            createBoard();
            setupHUD();
            showScreen('game');
            startTurn();
        }

        function createBoard() {
            const boardEl = document.getElementById('game-board');
            boardEl.innerHTML = '';
            board = [];

            // Tipos de casillas
            const types = ['normal', 'question', 'challenge', 'leak', 'superleak', 'joker'];
            const specialCount = Math.floor(50 * 0.4);

            // Casilla 0: Inicio
            board.push({ type: 'start', x: 0, y: 0 });
            const cell0 = createCell(0, 'start', 'INICIO');
            boardEl.appendChild(cell0);

            // Casillas 1 a 48
            for (let i = 1; i < 49; i++) {
                let type = 'normal';
                if (i < specialCount && Math.random() < 0.6) {
                    type = types[Math.floor(Math.random() * (types.length - 1)) + 1];
                }
                const x = i % 10;
                const y = Math.floor(i / 10);
                board.push({ type, x, y });
                const cell = createCell(i, type);
                boardEl.appendChild(cell);
            }

            // Casilla 49: Final
            board.push({ type: 'end', x: 9, y: 4 });
            const cell49 = createCell(49, 'end', 'JARRA');
            boardEl.appendChild(cell49);

            // Añadir tokens
            players.forEach(p => {
                const token = document.createElement('div');
                token.className = `player-token p${p.id + 1}`;
                token.style.background = p.color;
                token.id = `token-${p.id}`;
                getCellElement(0).appendChild(token);
            });
        }

        function createCell(index, type, text = '') {
            const cell = document.createElement('div');
            cell.className = `cell ${type}`;
            cell.dataset.index = index;
            if (text) cell.textContent = text;
            return cell;
        }

        function getCellElement(index) {
            return document.querySelector(`.cell[data-index="${index}"]`);
        }

        function setupHUD() {
            const hud = document.getElementById('hud');
            hud.innerHTML = '';
            players.forEach(p => {
                const div = document.createElement('div');
                div.className = 'player-info';
                div.innerHTML = `<h3 style="color:${p.color}">${p.name}</h3><div class="coins">💧 ${p.coins}</div>`;
                div.id = `info-${p.id}`;
                hud.appendChild(div);
            });
        }

        function updateHUD() {
            players.forEach(p => {
                const el = document.getElementById(`info-${p.id}`);
                if (el) el.querySelector('.coins').textContent = `💧 ${p.coins}`;
            });
        }

        function showScreen(screen) {
            Object.values(screens).forEach(s => s.classList.remove('active'));
            screens[screen].classList.add('active');
        }

        function startTurn() {
            const playerId = turnOrder[currentPlayerIndex];
            const player = players.find(p => p.id === playerId);
            document.getElementById('status').textContent = `Turno de ${player.name}. ¡Tira el dado!`;
            document.getElementById('roll-dice').style.display = 'inline-block';
            document.getElementById('roll-dice').onclick = rollDice;
        }

        function rollDice() {
            if (diceRolling) return;
            diceRolling = true;
            document.getElementById('roll-dice').style.display = 'none';
            const dice = document.getElementById('dice');
            dice.style.display = 'inline-flex';
            dice.textContent = '?';

            let rolls = 0;
            const interval = setInterval(() => {
                dice.textContent = Math.floor(Math.random() * 6) + 1;
                rolls++;
                if (rolls > 10) {
                    clearInterval(interval);
                    const result = Math.floor(Math.random() * 6) + 1;
                    dice.textContent = result;
                    setTimeout(() => movePlayer(result), 800);
                }
            }, 100);
        }

        function movePlayer(steps) {
            const playerId = turnOrder[currentPlayerIndex];
            const player = players.find(p => p.id === playerId);
            const oldPos = player.position;
            player.position = Math.min(oldPos + steps, 49);

            // Animar movimiento
            const token = document.getElementById(`token-${player.id}`);
            const targetCell = getCellElement(player.position);
            const rect = targetCell.getBoundingClientRect();
            const boardRect = document.getElementById('game-board').getBoundingClientRect();
            token.style.left = (rect.left - boardRect.left + 20) + 'px';
            token.style.top = (rect.top - boardRect.top + 10) + 'px';

            setTimeout(() => {
                diceRolling = false;
                document.getElementById('dice').style.display = 'none';
                handleCell(player.position);
            }, 600);
        }

        function handleCell(pos) {
            const cell = board[pos];
            const player = players.find(p => p.id === turnOrder[currentPlayerIndex]);

            if (pos === 49) {
                checkWinCondition();
                return;
            }

            if (cell.type === 'normal') {
                nextTurn();
                return;
            }

            if (player.joker && cell.type !== 'question' && cell.type !== 'challenge') {
                // Comodín protege de castigos
                player.joker = false;
                alert(`${player.name} usó su Comodín y evitó el castigo.`);
                nextTurn();
                return;
            }

            switch (cell.type) {
                case 'question':
                    showQuestion();
                    break;
                case 'challenge':
                    showChallenge();
                    break;
                case 'leak':
                    if (!player.joker) {
                        player.coins = Math.max(0, player.coins - 3);
                        updateHUD();
                        modals.leak.querySelector('p').innerHTML = `${player.name} perdió <strong>3 AquaCoins</strong>.`;
                        modals.leak.classList.add('active');
                        document.getElementById('close-leak').onclick = () => {
                            modals.leak.classList.remove('active');
                            nextTurn();
                        };
                    } else {
                        player.joker = false;
                        nextTurn();
                    }
                    break;
                case 'superleak':
                    showSuperLeak();
                    break;
                case 'joker':
                    player.joker = true;
                    modals.joker.classList.add('active');
                    document.getElementById('close-joker').onclick = () => {
                        modals.joker.classList.remove('active');
                        nextTurn();
                    };
                    break;
            }
        }

            opts.innerHTML = '';
            document.getElementById('question-feedback').textContent = '';

            q.a.forEach((ans, i) => {
                const btn = document.createElement('button');
                btn.className = 'option-btn';
                btn.textContent = ans;
                btn.onclick = () => {
                    document.querySelectorAll('.option-btn').forEach(b => b.disabled = true);
                    if (i === q.c) {
                        btn.classList.add('correct');
                        players.find(p => p.id === turnOrder[currentPlayerIndex]).coins++;
                        document.getElementById('question-feedback').textContent = '¡Correcto! +1 AquaCoin';
                    } else {
                        btn.classList.add('wrong');
                        const p = players.find(p => p.id === turnOrder[currentPlayerIndex]);
                        p.coins = Math.max(0, p.coins - 1);
                        document.getElementById('question-feedback').textContent = 'Incorrecto. -1 AquaCoin';
                    }
                    updateHUD();
                    setTimeout(() => {
                        modals.question.classList.remove('active');
                        nextTurn();
                    }, 2000);
                };
                opts.appendChild(btn);
            });

            modals.question.classList.add('active');
        }

        function showChallenge() {
            const challenge = CHALLENGES[Math.floor(Math.random() * CHALLENGES.length)];
            document.getElementById('challenge-title').textContent = '¡RETO!';
            document.getElementById('challenge-desc').textContent = challenge;
            document.getElementById('challenge-area').innerHTML = '<p>¡Tienen 3 minutos!</p>';
            
            let time = 180;
            const timer = document.getElementById('timer');
            timer.textContent = formatTime(time);
            const interval = setInterval(() => {
                time--;
                timer.textContent = formatTime(time);
                if (time <= 0) {
                    clearInterval(interval);
                    timer.textContent = '¡TIEMPO!';
                }
            }, 1000);

            document.getElementById('complete-challenge').onclick = () => {
                clearInterval(interval);
                players.find(p => p.id === turnOrder[currentPlayerIndex]).coins += 2;
                updateHUD();
                alert('¡Reto completado! +2 AquaCoins');
                modals.challenge.classList.remove('active');
                nextTurn();
            };

            modals.challenge.classList.add('active');
        }

        function showSuperLeak() {
            const segments = [
                { text: '+3 💧', value: 3, color: '#00aa00' },
                { text: '-2 💧', value: -2, color: '#aa0000' },
                { text: '+1 💧', value: 1, color: '#00cc00' },
                { text: '-3 💧', value: -3, color: '#cc0000' },
                { text: '+5 💧', value: 5, color: '#00ff00' },
                { text: '¡JOKER!', value: 'joker', color: '#ffcc00' }
            ];

            const roulette = document.getElementById('roulette');
            roulette.innerHTML = '';
            const angle = 360 / segments.length;

            segments.forEach((seg, i) => {
                const div = document.createElement('div');
                div.className = 'roulette-segment';
                div.style.background = seg.color;
                div.style.transform = `rotate(${i * angle}deg)`;
                div.textContent = seg.text;
                roulette.appendChild(div);
            });

            let spinning = false;
            document.getElementById('spin-roulette').onclick = () => {
                if (spinning) return;
                spinning = true;
                const spins = 5 + Math.random() * 5;
                const finalAngle = spins * 360 + Math.random() * 360;
                roulette.style.transform = `rotate(${finalAngle}deg)`;

                setTimeout(() => {
                    const normalized = finalAngle % 360;
                    const index = Math.floor((360 - normalized) / angle) % segments.length;
                    const result = segments[index];
                    document.getElementById('roulette-result').textContent = `¡${result.text}!`;

                    const player = players.find(p => p.id === turnOrder[currentPlayerIndex]);
                    if (result.value === 'joker') {
                        player.joker = true;
                    } else {
                        player.coins = Math.max(0, player.coins + result.value);
                    }
                    updateHUD();

                    setTimeout(() => {
                        modals.superleak.classList.remove('active');
                        nextTurn();
                    }, 2000);
                }, 3200);
            };

            modals.superleak.classList.add('active');
        }

        function nextTurn() {
            currentPlayerIndex = (currentPlayerIndex + 1) % turnOrder.length;
            setTimeout(startTurn, 800);
        }

        function checkWinCondition() {
            const player = players.find(p => p.id === turnOrder[currentPlayerIndex]);
            if (player.coins >= 10) {
                endGame(player);
            } else {
                document.getElementById('status').textContent = `${player.name} llegó al final, pero necesita 10+ AquaCoins. Sigue jugando...`;
                nextTurn();
            }
        }

        function endGame(winner) {
            // Conteo final
            let maxCoins = -1;
            let winners = [];
            players.forEach(p => {
                if (p.coins > maxCoins) {
                    maxCoins = p.coins;
                    winners = [p];
                } else if (p.coins === maxCoins) {
                    winners.push(p);
                }
            });

            if (winners.length === 1 && winners[0].id === winner.id) {
                document.getElementById('winner-name').textContent = `${winner.name} ¡GANÓ!`;
            } else {
                const names = winners.map(w => w.name).join(' y ');
                document.getElementById('winner-name').textContent = `${names} ¡EMPATE!`;
            }

            showScreen('winner');
        }

        function formatTime(seconds) {
            const m = Math.floor(seconds / 60);
            const s = seconds % 60;
            return `${m}:${s.toString().padStart(2, '0')}`;
        }
